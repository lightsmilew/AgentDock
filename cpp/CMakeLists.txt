cmake_minimum_required(VERSION 3.24)
project(AgentDock VERSION 1.0.0 LANGUAGES C CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "AgentDock native C++ requires Windows 10 1703 or later")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /W3 $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL> /Gy /Gw)
  add_link_options($<$<CONFIG:Release>:/LTCG> /OPT:REF /OPT:ICF /INCREMENTAL:NO)
endif()
add_library(sqlite STATIC vendor/sqlite3.c)
file(GLOB YAML_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/yaml/*.c")
add_library(yaml STATIC ${YAML_SOURCES})
target_include_directories(yaml PUBLIC vendor/yaml)
target_compile_definitions(yaml PUBLIC YAML_DECLARE_STATIC PRIVATE HAVE_CONFIG_H _CRT_SECURE_NO_WARNINGS)
target_compile_definitions(sqlite PRIVATE SQLITE_ENABLE_FTS5 SQLITE_OMIT_LOAD_EXTENSION SQLITE_OMIT_SHARED_CACHE SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_THREADSAFE=1)
add_library(agentdock_core STATIC src/util.cpp src/discovery.cpp src/cli_discovery.cpp src/agent_install.cpp src/official_install.cpp src/desktop_install.cpp src/removal.cpp src/compatibility.cpp src/db.cpp src/store.cpp src/models.cpp src/adapters.cpp src/adapter_extra.cpp src/harness_config.cpp src/zcode_config.cpp src/launch.cpp src/model_state.cpp)
target_include_directories(agentdock_core PUBLIC src vendor)
target_link_libraries(agentdock_core PUBLIC sqlite yaml bcrypt crypt32 winhttp ole32 shell32 advapi32 version)
# Explicit local header edges also support MSVC installations whose localized
# /showIncludes prefix is misdecoded by CMake/Ninja on a non-UTF-8 console.
file(GLOB LOCAL_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/*.h")
file(GLOB LOCAL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
set_source_files_properties(${LOCAL_SOURCES} PROPERTIES OBJECT_DEPENDS "${LOCAL_HEADERS}")
add_executable(AgentDock WIN32 src/main.cpp src/ui.cpp src/app.cpp src/app_background.cpp src/dialogs.cpp app.rc)
get_filename_component(AGENTDOCK_ASSET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../assets" ABSOLUTE)
file(GLOB BRAND_ASSETS CONFIGURE_DEPENDS "${AGENTDOCK_ASSET_DIR}/brands/*.svg" "${AGENTDOCK_ASSET_DIR}/brands/*.png" "${AGENTDOCK_ASSET_DIR}/brands/*.json" "${AGENTDOCK_ASSET_DIR}/catalog.json")
set_source_files_properties(app.rc PROPERTIES OBJECT_DEPENDS "${BRAND_ASSETS};${CMAKE_CURRENT_SOURCE_DIR}/app.manifest;${CMAKE_CURRENT_SOURCE_DIR}/../assets/icon.ico")
target_link_libraries(AgentDock PRIVATE agentdock_core d2d1 d3d11 dxgi dwrite dwmapi comctl32 shlwapi uxtheme windowscodecs)
target_link_options(AgentDock PRIVATE /MANIFEST:NO)
add_executable(discovery_fixture tests/discovery_fixture.cpp tests/discovery_fixture.rc)
add_executable(agentdock_tests tests/core_tests.cpp tests/desktop_install_tests.cpp tests/removal_tests.cpp tests/compatibility_tests.cpp tests/discovery_tests.cpp tests/network_tests.cpp tests/brand_tests.cpp tests/model_state_tests.cpp tests/adapter_tests.cpp app.rc)
add_dependencies(agentdock_tests discovery_fixture)
target_compile_definitions(agentdock_tests PRIVATE BRAND_SOURCE_FILE="${AGENTDOCK_ASSET_DIR}/brands/sources.json")
target_link_libraries(agentdock_tests PRIVATE agentdock_core ws2_32 d2d1 d3d11 dxgi shlwapi windowscodecs)
target_link_options(agentdock_tests PRIVATE /MANIFEST:NO)
enable_testing()
add_test(NAME native_core COMMAND agentdock_tests)
add_executable(asset_optimizer EXCLUDE_FROM_ALL tools/optimize_png.cpp)
target_link_libraries(asset_optimizer PRIVATE windowscodecs ole32)
